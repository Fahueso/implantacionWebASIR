DROP DATABASE IF EXISTS empresa;

CREATE DATABASE empresa
    CHARACTER SET utf8mb4
    COLLATE utf8mb4_unicode_ci;

USE empresa;

-- ------------------------------------------------------------
-- Tabla clientes (capítulos 3 a 6)
-- ------------------------------------------------------------
CREATE TABLE clientes (
    id       INT AUTO_INCREMENT PRIMARY KEY,
    nombre   VARCHAR(100) NOT NULL,
    telefono VARCHAR(20)
);

INSERT INTO clientes (nombre, telefono) VALUES

    ('Pedro', '600222222'),
    ('María', '600333333'),
    ('Marta', '600444444');

-- ------------------------------------------------------------
-- Tablas grupos y alumnos (capítulo 7)
-- ------------------------------------------------------------
CREATE TABLE grupos (
    id     INT AUTO_INCREMENT PRIMARY KEY,
    nombre VARCHAR(100) NOT NULL
); 

INSERT INTO grupos (nombre) VALUES
    ('ASIR'),
    ('DAW'),
    ('SMR');

CREATE TABLE alumnos (
    id       INT AUTO_INCREMENT PRIMARY KEY,
    nombre   VARCHAR(100) NOT NULL,
    grupo_id INT NULL,
    FOREIGN KEY (grupo_id) REFERENCES grupos(id)
);

-- ------------------------------------------------------------
-- Usuario de la aplicación (capítulos 2 y 8)
-- Mínimo privilegio y restringido a localhost.
-- ¡Cambia la contraseña en un entorno real!
-- ------------------------------------------------------------
DROP USER IF EXISTS 'app_web'@'localhost';

CREATE USER 'app_web'@'localhost'
    IDENTIFIED BY 'password_seguro';

GRANT SELECT, INSERT, UPDATE, DELETE
    ON empresa.*
    TO 'app_web'@'localhost';
